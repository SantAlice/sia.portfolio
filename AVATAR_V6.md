# Аватар в шапке — дорожка v6

Заменяет все предыдущие задания по аватару (v2–v5). Проект — статический сайт (`index.html`, `css/`, `js/`), не React.

Отличие от v5: дуга стала почти полной — покрыто 290° из 360°. Непокрыт только верх. Логика та же (дуга + возврат в нейтраль), но `MAP`, длительность, цвет фона и точка отсчёта — другие. Старые значения не подойдут.

## Состояние покоя

`avatar-poster-v6.jpg` — ровно первый кадр дорожки: фронтальная нейтральная поза, взгляд в камеру. До первого движения мыши постер и первый кадр видео совпадают пиксель в пиксель, подмены при загрузке не видно.

---

## Как это работает

Дорожка идёт так: нейтраль → взгляд вверх-вправо → вправо → вниз-вправо → вниз → вниз-влево → влево → вверх-влево → обратно в нейтраль.

Покрытая дуга — **от 315° через 0° до 245°** в экранных градусах, где 0 вправо, 90 вниз, 180 влево, 270 вверх. Не покрыт только сектор **245°–315°** — верх.

Когда курсор попадает в этот сектор, взгляд возвращается в нейтраль. Причём возврат идёт по реальным кадрам: из левого края дуги — через кадры 167–212 (персонаж сам опускает голову к фронту), из правого — через кадры 24–0 в обратную сторону. Это не заглушка и не кроссфейд, а настоящая анимация из исходника.

Расположение: **аватар справа, текст слева**. Курсор тогда почти всегда слева-снизу от лица, то есть в самой плотной части дуги.

---

## Стык

Первый и последний кадр дорожки — одна и та же фронтальная нейтральная поза, расхождение между ними меньше половины межкадрового шага. Переход через край дорожки не виден. Кратчайшая дуга в `loop()` этим и пользуется — не убирать её.

## JS

```js
(function () {
  var box = document.getElementById('avatarBox');
  var video = document.getElementById('avatarVideo');
  if (!box || !video) return;

  var DURATION = 8.875;    // длительность дорожки, секунды
  var EASE = 0.1;          // инерция: меньше — взгляд догоняет ленивее
  var AR = 720 / 1076;     // пропорция видео

  // Покрытая дуга: направление взгляда -> позиция на дорожке 0..1.
  // Градусы экранные: 0 вправо, 90 вниз, 180 влево, 270 вверх.
  // Дуга проходит через 0°, поэтому углы развёрнуты в непрерывную
  // шкалу 315..605 (605 = 245 + 360). Значения сняты покадрово по
  // положению лица относительно головы во всех 213 кадрах —
  // скорость движения по дуге неравномерная, от 1 до 7 градусов
  // на кадр. Формулой не выводится, пересчитывать не нужно.
  var MAP = [
    [ 315.0, 0.1150 ], [ 328.0, 0.1432 ], [ 336.0, 0.1573 ],
    [ 342.0, 0.1714 ], [ 350.0, 0.1854 ], [ 358.0, 0.1995 ],
    [ 361.0, 0.2089 ], [ 378.0, 0.2559 ], [ 384.0, 0.2700 ],
    [ 392.0, 0.2840 ], [ 405.0, 0.3122 ], [ 415.0, 0.3404 ],
    [ 425.0, 0.3685 ], [ 437.0, 0.3967 ], [ 448.0, 0.4249 ],
    [ 456.0, 0.4531 ], [ 463.0, 0.4859 ], [ 468.0, 0.5469 ],
    [ 468.5, 0.5516 ], [ 490.0, 0.5704 ], [ 505.0, 0.5939 ],
    [ 532.0, 0.6127 ], [ 548.0, 0.6362 ], [ 560.0, 0.6596 ],
    [ 574.0, 0.6831 ], [ 584.0, 0.7066 ], [ 593.0, 0.7254 ],
    [ 600.0, 0.7488 ], [ 605.0, 0.7817 ]
  ];

  var ARC_LO = 315;        // начало дуги — взгляд вверх-вправо
  var ARC_HI = 605;        // конец дуги — взгляд вверх-влево (245° + 360)
  var NEUTRAL = 0.0;       // фронтальная поза, первый кадр дорожки

  function angleToPos(deg) {
    // разворачиваем в непрерывную шкалу: дуга идёт через 0°
    var u = deg % 360;
    if (u < 0) u += 360;
    if (u < 245) u += 360;

    if (u >= ARC_LO && u <= ARC_HI) {
      for (var i = 0; i < MAP.length - 1; i++) {
        var a = MAP[i], b = MAP[i + 1];
        if (u >= a[0] && u <= b[0]) {
          return a[1] + (b[1] - a[1]) * ((u - a[0]) / (b[0] - a[0]));
        }
      }
    }
    // сектор 245..315 — верх, в дорожке его нет: идём в нейтраль.
    // К какому её краю — решает кратчайшая дуга в loop(),
    // поэтому голова возвращается той же стороной, откуда пришла.
    return NEUTRAL;
  }

  // точка отсчёта — глаза персонажа внутри РЕАЛЬНОЙ области видео,
  // а не внутри блока: при object-fit: contain блок может быть больше
  // видео, и отсчёт от блока даёт систематическую ошибку угла
  function eyePoint() {
    var r = box.getBoundingClientRect();
    var vw, vh;
    if (r.width / r.height > AR) { vh = r.height; vw = vh * AR; }
    else                         { vw = r.width;  vh = vw / AR; }
    return {
      x: r.left + (r.width  - vw) / 2 + vw * 0.482,
      y: r.top  + (r.height - vh) / 2 + vh * 0.184
    };
  }

  var still = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var canHover = window.matchMedia('(hover: hover)').matches;

  // на тач-устройствах курсора нет — крутим дорожку в цикле
  if (!canHover || still) {
    video.loop = true;
    video.play().catch(function () {});
    return;
  }

  video.pause();

  var target = NEUTRAL;
  var current = NEUTRAL;

  video.addEventListener('loadedmetadata', function () {
    video.currentTime = 0;
  }, { once: true });

  window.addEventListener('mousemove', function (e) {
    var p = eyePoint();
    var deg = Math.atan2(e.clientY - p.y, e.clientX - p.x) * 180 / Math.PI;
    if (deg < 0) deg += 360;
    target = angleToPos(deg);
  }, { passive: true });

  function loop() {
    // кратчайшая дуга: оба конца дорожки — одна и та же нейтральная
    // поза, поэтому переход через край непрерывен. Не убирать.
    var diff = target - current;
    if (diff > 0.5) diff -= 1;
    if (diff < -0.5) diff += 1;
    current = (current + diff * EASE + 1) % 1;

    if (video.readyState >= 1) {
      var t = current * DURATION;
      if (isFinite(t)) {
        video.currentTime = Math.min(DURATION - 0.001, Math.max(0, t));
      }
    }

    requestAnimationFrame(loop);
  }
  requestAnimationFrame(loop);
})();
```

## Настройка после запуска

| Симптом | Что менять |
|---|---|
| Взгляд систематически не доходит по всей дуге | левая колонка `MAP` — сдвинуть все углы на одну величину |
| Промах в одном секторе | соседние узлы `MAP` для этого сектора |
| Промах зависит от близости курсора к аватару | `0.482` и `0.184` в `eyePoint()` |
| Возврат в нейтраль слишком резкий | `EASE` уменьшить до 0.06 |
| Догоняет слишком вяло | `EASE` увеличить до 0.18 |

Менять по одному значению за раз.

---

## Что в дорожке есть по-честному, а что нет

- **Есть:** вправо, вверх-вправо, вниз-вправо, вниз, вниз-влево, влево, вверх-влево. Это 290° из 360°.
- **Нет:** сектор 245°–315°, то есть строго верх. Там нейтраль.
- **Моргание.** В районе 5°–18° (курсор чуть ниже-правее лица) персонаж моргает — в исходнике там закрытые глаза, 11 кадров. Читается как обычное моргание, специально не чинится. Если будет мешать: в `MAP` заменить пару `[361, 0.2089], [378, 0.2559]` на `[361, 0.2089], [362, 0.2582]` — тогда закрытые кадры сожмутся в 1° и глаза будут открыты почти везде, ценой резкого скачка головы на этой границе.

## Проверка перед сдачей

1. Курсор строго слева от лица → профиль влево
2. Курсор строго снизу → голова опущена
3. Курсор строго справа → голова повёрнута вправо, глаза открыты
4. Курсор строго сверху → персонаж плавно уходит в нейтраль, смотрит вперёд, без рывка
5. Медленно провести курсором по кругу вокруг аватара — голова идёт за ним без разворотов «через всю голову»
6. Границ прямоугольника видео не видно ни в одном кадре
7. Медленно сжать окно от широкого до узкого — вёрстка не ломается
8. На узком экране колонки встают в стопку, аватар сверху
9. На телефоне дорожка крутится сама по себе в цикле

## Чего делать не нужно

- Не ставить `autoplay` и `loop` на десктопе
- Не пережимать `avatar-orbit-v6.mp4` — в нём уплотнены keyframe, выровнен фон и вырезана пауза; кадр не обрезан — гриб входит целиком
- Не «дополнять» `MAP` значениями для верхнего сектора — их в дорожке нет
- Не убирать кратчайшую дугу из `loop()`
- Не убирать разворот `if (u < 245) u += 360` — без него дуга через 0° разваливается
- Не пересчитывать `MAP` формулой
- Не считать точку отсчёта от `getBoundingClientRect()` напрямую — только через `eyePoint()`
- Не менять `aspect-ratio`
- Не задавать `min-height: 100vh` для `.hero`
- Не накладывать текст поверх аватара — фон видео непрозрачный
